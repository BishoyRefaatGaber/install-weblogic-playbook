define these variables 

    ORACLE BASE :: Default Oracle installer directory location
    ORACLE HOME :: Default Oracle database directory location / optional if have Oracle client inside
    MW HOME :: Default Middleware installer directory location
    WLS HOME :: Default Oracle Weblogic managed server directory location
    WL HOME :: Default Oracle Weblogic admin server directory location
    DOMAIN BASE :: Default Oracle Weblogic global domain
    DOMAIN HOME :: Default Oracle Weblogic specific domain

# Start WLST

# Configure Admin Server
wls:/offline/base_domain> cd('Servers/AdminServer')
wls:/offline/base_domain/Server/AdminServer> set('ListenAddress','<server-IP>')  # Fixed attribute name
wls:/offline/base_domain/Server/AdminServer> set('ListenPort',7001)

# Create SSL Configuration
wls:/offline/base_domain/Server/AdminServer> create('AdminServer','SSL')  # Create SSL MBean
wls:/offline/base_domain/Server/AdminServer> cd('SSL/AdminServer')
wls:/offline/base_domain/Server/AdminServer/SSL/AdminServer> set('Enabled','true')  # Use lowercase 'true'
wls:/offline/base_domain/Server/AdminServer/SSL/AdminServer> set('ListenPort',7002)

# Configure Security
wls:/offline/base_domain/Server/AdminServer/SSL/AdminServer> cd('/')

wls:/offline/base_domain> cd('Security/base_domain/User/weblogic')  # Correct security path
wls:/offline/base_domain/Security/base_domain/User/weblogic> cmo.setPassword('Test1234')

# Finalize Domain
wls:/offline/base_domain/Security/base_domain/User/weblogic> setOption('OverwriteDomain','true')
wls:/offline/base_domain> writeDomain('/home/weblogic/wls/oracle/config/domains/TEST')  # Use DOMAIN_HOME variable

# Cleanup
wls:/offline/TEST/Security/TEST/User/weblogic> closeTemplate()
wls:/offline> exit()
